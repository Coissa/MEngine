file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tmp")
function(engine_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE Engine)
    target_compile_options(${name} PRIVATE /EHsc /utf-8)
    target_include_directories(${name} PRIVATE "${PROJECT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/Smoke")
    target_compile_definitions(${name} PRIVATE MENGINE_TEST_ASSET_ROOT="${PROJECT_SOURCE_DIR}/Assets")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30 ENVIRONMENT "TEMP=${CMAKE_CURRENT_BINARY_DIR}/tmp;TMP=${CMAKE_CURRENT_BINARY_DIR}/tmp")
endfunction()
file(GLOB smoke_tests "${CMAKE_CURRENT_SOURCE_DIR}/Smoke/*.cpp")
foreach(source IN LISTS smoke_tests)
    get_filename_component(name "${source}" NAME_WE)
    engine_test(${name} "${source}")
endforeach()
